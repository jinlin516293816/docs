4. 事件总线（发布订阅模式） ​

适用于非父子组件间的通信。

jsx
const eventBus = {
  events: {},
  on(event, callback) {
    if (!this.events[event]) {
      this.events[event] = [];
    }
    this.events[event].push(callback);
  },
  emit(event, data) {
    if (this.events[event]) {
      this.events[event].forEach(callback => callback(data));
    }
  }
};

// 组件A订阅
eventBus.on('userUpdate', (user) => console.log(user));

// 组件B发布
eventBus.emit('userUpdate', { name: 'John' });

3. 自定义Hook封装 ​

将可复用逻辑封装为自定义Hook。

jsx
function useLocalStorage(key, initialValue) {
  const [storedValue, setStoredValue] = useState(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch {
      return initialValue;
    }
  });

  const setValue = (value) => {
    try {
      const valueToStore = value instanceof Function ? value(storedValue) : value;
      setStoredValue(valueToStore);
      window.localStorage.setItem(key, JSON.stringify(valueToStore));
    } catch (error) {
      console.error(error);
    }
  };

  return [storedValue, setValue];
}

// 使用
const [theme, setTheme] = useLocalStorage('theme', 'light');

9. React中的setState同步还是异步？ ​

React 18之前：

  • 在Promise的状态更新、JS原生事件、setTimeout、setInterval中是同步的
  • 在React的合成事件中，是异步的

setState的"异步"并不是说内部由异步代码实现，其实本身执行的过程和代码都是同步的，只是合成事件和钩子函数的调用顺序在更新之前，导致在合成事件和钩子函数中没法立马拿到更新后的值。

React 18之后： setState都会表现为异步（批处理）。

jsx
// React 18之前
class Counter extends React.Component {
  state = { count: 0 };

  handleClick() {
    // 合成事件中 - 异步
    this.setState({ count: this.state.count + 1 });
    console.log(this.state.count); // 0 - 打印的还是旧值

    // 回调函数中获取新值
    this.setState({ count: this.state.count + 1 }, () => {
      console.log(this.state.count); // 2 - 回调中拿到新值
    });
  }

  componentDidMount() {
    // 原生事件中 - 同步
    document.body.addEventListener('click', () => {
      this.setState({ count: this.state.count + 1 });
      console.log(this.state.count); // 最新值
    });
  }
}

// React 18 - 全部异步批处理
function Counter() {
  const [count, setCount] = useState(0);

  const handleClick = () => {
    // 批处理 - 不会立即更新
    setCount(c => c + 1);
    console.log(count); // 仍然是旧值

    setCount(c => c + 1);
    console.log(count); // 仍然旧值

    // 最终只执行一次渲染，count变为2
  };

  return <button onClick={handleClick}>{count}</button>;
}

10. 虚拟DOM的工作原理 ​

虚拟DOM是内存中的DOM表示。React使用虚拟DOM来优化DOM的更新过程。

三个步骤：

  1. 状态更新：当组件状态发生变化时，React创建新的虚拟DOM树
  2. 差异对比：React比较新旧虚拟DOM的差异（Diff算法）
  3. 最小更新：只将变化的部分更新到真实DOM
jsx
// 虚拟DOM示例
const virtualDom = {
  type: 'div',
  props: { className: 'container' },
  children: [
    {

13. useEffect与useLayoutEffect的区别 ​

useEffect在React的渲染过程中被异步调用，适用于绝大多数场景。 useLayoutEffect会在所有的DOM变更之后同步调用，用于处理DOM操作、调整样式等。

jsx
function Component() {
  // useEffect - 渲染后异步执行，可能闪烁
  useEffect(() => {
    document.title = '标题已更新';
    // 用于数据获取、订阅等
  }, [依赖]);

  // useLayoutEffect - 渲染前同步执行，不闪烁
  useLayoutEffect(() => {
    // 获取DOM尺寸/位置
    const rect = elementRef.current.getBoundingClientRect();
    console.log(rect);

    // 同步修改样式，避免闪烁
    elementRef.current.style.transform = 'translateX(100px)';
  }, []);

  return <div ref={elementRef}>内容</div>;
}

// 执行顺序
function Parent() {

function TodoList() {
  const [state, dispatch] = useReducer(todoReducer, {
    todos: [],
    filter: 'all',
    loading: false
  });
  
  // 集中管理状态逻辑
  const addTodo = (text) => dispatch({ type: 'ADD_TODO', payload: text });
  const toggleTodo = (id) => dispatch({ type: 'TOGGLE_TODO', payload: id });
  const deleteTodo = (id) => dispatch({ type: 'DELETE_TODO', payload: id });
  const setFilter = (filter) => dispatch({ type: 'SET_FILTER', payload: filter });
  
  return (
    <div>
      <input onKeyDown={(e) => e.key === 'Enter' && addTodo(e.target.value)} />
      <ul>
        {state.todos
          .filter(todo => 
            state.filter === 'all' || 
            (state.filter === 'active' && !todo.completed) ||
            (state.filter === 'completed' && todo.completed)
          )
          .map(todo => (
            <li
              key={todo.id}
              onClick={() => toggleTodo(todo.id)}
              style={{ textDecoration: todo.completed ? 'line-through' : 'none' }}
            >
              {todo.text}
              <button onClick={() => deleteTodo(todo.id)}>删除</button>
            </li>
          ))}
      </ul>
      <div>
        <button onClick={() => setFilter('all')}>全部</button>
        <button onClick={() => setFilter('active')}>未完成</button>
        <button onClick={() => setFilter('completed')}>已完成</button>
      </div>
    </div>
  );
}

// Reducer函数
function todoReducer(state, action) {
  switch (action.type) {
    case 'ADD_TODO':
      return {
        ...state,
        todos: [
          ...state.todos,
          { id: Date.now(), text: action.payload, completed: false }
        ]
      };
    case 'TOGGLE_TODO':
      return {
        ...state,
        todos: state.todos.map(todo =>
          todo.id === action.payload
            ? { ...todo, completed: !todo.completed }
            : todo
        )
      };
    case 'DELETE_TODO':
      return {
        ...state,
        todos: state.todos.filter(todo => todo.id !== action.payload)
      };
    case 'SET_FILTER':
      return { ...state, filter: action.payload };
    default:
      return state;
  }
}
