3. never 类型的使用场景？ ​

答案：

never 类型表示永远不会发生的值的类型，主要用于：

typescript
// 1. 抛出异常的函数
function throwError(msg: string): never {
  throw new Error(msg);
}

// 2. 无限循环
function infiniteLoop(): never {
  while (true) {}
}

// 3. 类型守卫中穷尽检查
type Fruit = 'apple' | 'banana';
function getFruitColor(fruit: Fruit): string {
  switch (fruit) {
    case 'apple': return 'red';
    case 'banana': return 'yellow';
    // 如果漏掉 case，TypeScript 会报错
    default: return neverReached(fruit); // fruit 的类型是 never
  }
}

function neverReached(value: never): never {
  throw new Error(`Unexpected value: ${value}`);
}

四、泛型 ​

8. 什么是泛型？为什么需要泛型？ ​

答案：

泛型允许在定义函数、接口或类时不预先指定具体类型，而是在使用时再指定。

好处：

  • 类型安全：编译时检查类型
  • 代码复用：一套代码支持多种类型
  • 更好的 IDE 支持：智能提示
typescript
// 不使用泛型 - 需要为每种类型写重复代码
function reverseString(arr: string[]): string[] {
  return arr.reverse();
}

function reverseNumber(arr: number[]): number[] {
  return arr.reverse();
}

// 使用泛型 - 一次定义，多次使用
function reverse<T>(arr: T[]): T[] {
  return arr.reverse();
}

const strs = reverse(['a', 'b', 'c']); // string[]
const nums = reverse([1, 2, 3]); // number[]

9. 泛型约束的方式？ ​

答案：

typescript
// 方式1：使用接口约束
interface Lengthwise {
  length: number;
}

function logLength<T extends Lengthwise>(arg: T): T {
  console.log(arg.length);
  return arg;

logLength([1, 2, 3]); // OK
// logLength(123); // 错误：number 没有 length 属性

// 方式2：使用 keyof 约束对象属性
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key];

const user = { name: 'Alice', age: 25 };
getProperty(user, 'name'); // string
getProperty(user, 'age'); // number
// getProperty(user, 'email'); // 错误：'email' 不是 user 的属性

六、装饰器 ​

15. 装饰器的作用和类型？ ​

答案：

装饰器是一种特殊类型的声明，可以附加到类声明、方法、访问器、属性或参数上。

装饰器类型：

typescript
// 1. 类装饰器
function sealed(constructor: Function) {
  Object.seal(constructor);
  Object.seal(constructor.prototype);
}

@sealed
class Greeter {
  greeting: string;
  constructor(message: string) {
    this.greeting = message;
  }
}

// 2. 方法装饰器
function enumerable(value: boolean) {
  return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    descriptor.enumerable = value;
  };
}

class Greeter {
  @enumerable(false)
  greet() {
    return 'Hello';
  }
}

// 3. 属性装饰器
function format(formatString: string) {
  return function (target: any, propertyKey: string) {
    // ...
  };
}

class Greeter {
  @format('Hello, %s')
  greeting: string;
}

// 4. 参数装饰器
function log(target: any, propertyKey: string, parameterIndex: number) {
  console.log(`参数索引: ${parameterIndex}`);
}

class Greeter {
  greet(@log message: string) {
    return message;
  }
}

十、综合面试题 ​

22. 如何实现类型安全的事件发射器？ ​

答案：

typescript
interface Events {
  'click': { x: number; y: number };
  'scroll': { top: number };
  'focus': undefined;
}

class EventEmitter<T extends Record<string, any>> {
  private listeners: {
    [K in keyof T]?: Array<(args: T[K]) => void>;
  } = {};

  on<K extends keyof T>(event: K, listener: (args: T[K]) => void) {
    if (!this.listeners[event]) {
      this.listeners[event] = [];
    }
    this.listeners[event]!.push(listener);
  }

  emit<K extends keyof T>(event: K, args: T[K]) {
    this.listeners[event]?.forEach(listener => listener(args));
  }
}

const emitter = new EventEmitter<Events>();
emitter.on('click', (args) => console.log(args.x, args.y));
emitter.emit('click', { x: 10, y: 20 }); // 正确
// emitter.emit('click', { x: 10 }); // 错误：缺少 y

type Action<T extends string, P = undefined> = P extends undefined
  ? { type: T }

type CounterActions =
  | Action<'INCREMENT'>
  | Action<'DECREMENT'>
  | Action<'SET', number>;

// Reducer
function counterReducer(state: number = 0, action: CounterActions): number {
  switch (action.type) {
    case 'INCREMENT':
      return state + 1;
    case 'DECREMENT':
      return state - 1;
    case 'SET':
      return action.payload; // 类型安全
    default:
      return state;
  }
}

24. 如何在 TypeScript 中处理异步操作？ ​

typescript
// 使用 Promise
async function fetchData(): Promise<User> {
  const response = await fetch('/api/user');
  return response.json();

// 使用泛型封装
type AsyncResult<T> =
  | { status: 'loading' }
  | { status: 'success'; data: T }
  | { status: 'error'; error: Error };

async function fetchAsync<T>(url: string): Promise<AsyncResult<T>> {
  try {
    const response = await fetch(url);
    const data = await response.json();
    return { status: 'success', data };
  } catch (error) {
    return { status: 'error', error: error as Error };
  }
}

// 使用

if (result.status === 'success') {
  console.log(result.data.name); // 类型安全
}

27. 如何实现类型安全的表单验证？ ​

答案：

typescript
interface FormValues {
  username: string;
  email: string;
  password: string;
}

interface FormErrors<T> {
  [K in keyof T]?: string;
}

function validateForm(values: FormValues): FormErrors<FormValues> {
  const errors: FormErrors<FormValues> = {};
  
  if (!values.username.trim()) {
    errors.username = '用户名不能为空';
  }
  
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(values.email)) {
    errors.email = '请输入有效的邮箱';
  }
  
  if (values.password.length < 6) {
    errors.password = '密码至少需要6位';
  }
  
  return errors;
}

// 使用
const errors = validateForm({
  username: '',
  email: 'invalid',
  password: '123'
});

errors.username; // string | undefined
errors.email;    // string | undefined

28. 如何实现类型安全的状态机？ ​

答案：

typescript
type State = 'idle' | 'loading' | 'success' | 'error';
type Event = 'fetch' | 'success' | 'error' | 'reset';

type Transition = {
  idle: { fetch: 'loading' };
  loading: { success: 'success'; error: 'error' };
  success: { fetch: 'loading'; reset: 'idle' };
  error: { fetch: 'loading'; reset: 'idle' };
};

class StateMachine<S extends State> {
  private state: S;
  
  constructor(initialState: S) {
    this.state = initialState;
  }
  
  transition<E extends keyof Transition[S]>(event: E): void {
    this.state = Transition[S][E] as S;
  }
  
  getState(): S {
    return this.state;
  }
}

const machine = new StateMachine<'idle'>('idle');
machine.transition('fetch'); // state: 'loading'
machine.transition('success'); // state: 'success'
// machine.transition('fetch'); // 错误：success 状态没有 fetch 事件

十三、总结 ​

TypeScript 面试重点：

  1. 基础类型：理解各类型特点，特别是 any、unknown、never
  2. 类型系统：接口、类型别名、泛型的使用和区别
  3. 高级类型：交叉、联合、条件、映射类型
  4. 类型体操：实现常用工具类型
  5. 实战应用：类型安全的状态管理、表单验证、事件系统
  6. 配置优化：tsconfig.json 关键配置

掌握这些知识，可以应对大部分 TypeScript 面试题！
