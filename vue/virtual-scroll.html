可视化区域虚拟加载计算逻辑 ​

什么是虚拟滚动？ ​

虚拟滚动（Virtual Scrolling）是一种性能优化技术，主要用于处理大量数据列表的渲染。它的核心思想是：只渲染可视区域内的元素，而不是渲染所有数据项。

为什么需要虚拟滚动？ ​

当列表数据量很大时（如成千上万条），如果一次性渲染所有 DOM 节点会导致：

  • 内存占用过高：大量 DOM 节点占用内存
  • 渲染性能差：浏览器需要计算和绘制大量元素
  • 页面卡顿：滚动时 FPS 下降，用户体验差

虚拟滚动通过只渲染可见区域的元素，将 DOM 节点数量控制在合理范围内，从而大幅提升性能。

2. 计算可视区域的项目范围 ​

javascript
/**
 * 计算当前滚动位置下，应该渲染哪些项目
 */
function getVisibleRange(scrollTop, itemHeight, containerHeight, totalItems) {
  // 计算起始索引（向下取整）
  const startIndex = Math.floor(scrollTop / itemHeight);
  
  // 计算可视区域内能显示的项目数量（向上取整，确保覆盖整个可视区域）
  const visibleCount = Math.ceil(containerHeight / itemHeight);
  
  // 计算结束索引（不超过总数量）
  const endIndex = Math.min(startIndex + visibleCount, totalItems);
  
  return {
    startIndex,
    endIndex,
    visibleCount
  };
}

3. 计算占位空间 ​

为了让滚动条正确反映总数据量，需要在列表前后添加占位元素：

javascript
/**
 * 计算前后占位高度
 */
function getSpacer Heights(startIndex, endIndex, itemHeight, totalItems) {
  // 前方占位高度（已滚动过的区域）
  const topSpacerHeight = startIndex * itemHeight;
  
  // 后方占位高度（还未滚动的区域）
  const bottomSpacerHeight = (totalItems - endIndex) * itemHeight;
  
  return {
    topSpacerHeight,
    bottomSpacerHeight
  };
}

4. 完整的虚拟滚动计算示例 ​

javascript
class VirtualScroller {
  constructor(options) {
    this.itemHeight = options.itemHeight || 50;
    this.containerHeight = options.containerHeight || 600;
    this.totalItems = options.totalItems || 0;
    this.data = options.data || [];
    
    // 当前状态
    this.scrollTop = 0;
    this.visibleData = [];
    this.offset = 0;
  }
  
  /**
   * 根据滚动位置更新可视数据
   */
  update(scrollTop) {
    this.scrollTop = scrollTop;
    
    // 计算可视范围
    const { startIndex, endIndex } = this.getVisibleRange();
    
    // 获取可视区域的数据
    this.visibleData = this.data.slice(startIndex, endIndex);
    
    // 计算偏移量（用于 CSS transform 或 margin-top）
    this.offset = startIndex * this.itemHeight;
    
    return {
      visibleData: this.visibleData,
      offset: this.offset,
      startIndex,
      endIndex
    };
  }
  
  /**
   * 计算可视范围
   */
  getVisibleRange() {
    const startIndex = Math.floor(this.scrollTop / this.itemHeight);
    const visibleCount = Math.ceil(this.containerHeight / this.itemHeight);
    const endIndex = Math.min(startIndex + visibleCount, this.totalItems);
    
    return { startIndex, endIndex };
  }
  
  /**
   * 计算总高度（用于设置容器滚动高度）
   */
  getTotalHeight() {
    return this.totalItems * this.itemHeight;
  }
}

Vue 3 实现示例 ​

基础版本（固定高度） ​

vue
<template>
  <div 
    class="virtual-list-container" 
    :style="{ height: containerHeight + 'px' }"
    @scroll="onScroll"
  >
    <!-- 内部滚动容器，设置总高度 -->
    <div 
      class="virtual-list-content" 
      :style="{ height: totalHeight + 'px' }"
    >
      <!-- 上方占位 -->
      <div :style="{ height: offset + 'px' }"></div>
      
      <!-- 可视区域的项目 -->
      <div
        v-for="(item, index) in visibleData"
        :key="startIndex + index"
        class="list-item"
        :style="{ height: itemHeight + 'px' }"
      >
        {{ item.label }} - Index: {{ startIndex + index }}
      </div>
      
      <!-- 下方占位 -->
      <div :style="{ height: bottomSpacerHeight + 'px' }"></div>
    </div>
  </div>
</template>

<script setup>
import { ref, computed, onMounted } from 'vue';

const props = defineProps({
  data: {
    type: Array,
    required: true
  },
  itemHeight: {
    type: Number,
    default: 50
  },
  containerHeight: {
    type: Number,
    default: 600
  }
});

const scrollTop = ref(0);
const startIndex = ref(0);
const endIndex = ref(0);

// 计算属性
const totalItems = computed(() => props.data.length);
const totalHeight = computed(() => totalItems.value * props.itemHeight);

const visibleCount = computed(() => {
  return Math.ceil(props.containerHeight / props.itemHeight);
});

const visibleData = computed(() => {
  return props.data.slice(startIndex.value, endIndex.value);
});

const offset = computed(() => {
  return startIndex.value * props.itemHeight;
});

const bottomSpacerHeight = computed(() => {
  return (totalItems.value - endIndex.value) * props.itemHeight;
});

// 滚动处理
function onScroll(event) {
  scrollTop.value = event.target.scrollTop;
  updateVisibleRange();
}

// 更新可视范围
function updateVisibleRange() {
  startIndex.value = Math.floor(scrollTop.value / props.itemHeight);
  endIndex.value = Math.min(
    startIndex.value + visibleCount.value + 1, // +1 作为缓冲
    totalItems.value
  );
}

// 初始化
onMounted(() => {
  updateVisibleRange();
});
</script>

动态高度支持 ​

对于不定高度的项目，需要使用更复杂的策略：

方案 1：预估高度 + 实际测量 ​

javascript
class DynamicVirtualScroller {
  constructor(options) {
    this.estimatedItemHeight = options.estimatedItemHeight || 50;
    this.containerHeight = options.containerHeight || 600;
    this.data = options.data || [];
    
    // 存储每个项目的实际高度
    this.heightCache = new Map();
    // 存储每个项目的累积偏移量
    this.offsetCache = new Map();
    
    this.scrollTop = 0;
  }
  
  /**
   * 根据索引获取项目的累积偏移量
   */
  getOffset(index) {
    if (this.offsetCache.has(index)) {
      return this.offsetCache.get(index);
    }
    
    let offset = 0;
    for (let i = 0; i < index; i++) {
      offset += this.getItemHeight(i);
    }
    
    this.offsetCache.set(index, offset);
    return offset;
  }
  
  /**
   * 获取项目的实际高度（如果未测量则使用预估值）
   */
  getItemHeight(index) {
    return this.heightCache.get(index) || this.estimatedItemHeight;
  }
  
  /**
   * 更新项目的实际高度
   */
  updateItemHeight(index, height) {
    const oldHeight = this.heightCache.get(index) || this.estimatedItemHeight;
    this.heightCache.set(index, height);
    
    // 清除后续项目的缓存（因为偏移量改变了）
    for (let i = index + 1; i < this.data.length; i++) {
      this.offsetCache.delete(i);
    }
  }
  
  /**
   * 根据滚动位置找到起始索引
   */
  findStartIndex(scrollTop) {
    let low = 0;
    let high = this.data.length - 1;
    
    while (low <= high) {
      const mid = Math.floor((low + high) / 2);
      const offset = this.getOffset(mid);
      
      if (offset === scrollTop) {
        return mid;
      } else if (offset < scrollTop) {
        low = mid + 1;
      } else {
        high = mid - 1;
      }
    }
    
    return low;
  }
  
  /**
   * 计算可视范围
   */
  getVisibleRange() {
    const startIndex = this.findStartIndex(this.scrollTop);
    
    let endIndex = startIndex;
    let accumulatedHeight = 0;
    
    // 累加高度直到超过容器高度
    while (accumulatedHeight < this.containerHeight && endIndex < this.data.length) {
      accumulatedHeight += this.getItemHeight(endIndex);
      endIndex++;
    }
    
    return { startIndex, endIndex };
  }
}

Vue 3 动态高度实现 ​

vue
<template>
  <div 
    class="virtual-list-container" 
    :style="{ height: containerHeight + 'px' }"
    @scroll="onScroll"
  >
    <div 
      class="virtual-list-content" 
      :style="{ height: totalHeight + 'px' }"
    >
      <div :style="{ height: offset + 'px' }"></div>
      
      <div
        v-for="(item, index) in visibleData"
        :key="startIndex + index"
        class="list-item"
        ref="itemRefs"
        @resize="onItemResize(startIndex + index, $event)"
      >
        <component :is="item.component" :data="item" />
      </div>
      
      <div :style="{ height: bottomSpacerHeight + 'px' }"></div>
    </div>
  </div>
</template>

<script setup>
import { ref, computed, nextTick, onUpdated } from 'vue';

const props = defineProps({
  data: Array,
  estimatedItemHeight: {
    type: Number,
    default: 50
  },
  containerHeight: {
    type: Number,
    default: 600
  }
});

const scrollTop = ref(0);
const startIndex = ref(0);
const endIndex = ref(0);
const heightCache = ref(new Map());
const itemRefs = ref([]);

const totalHeight = computed(() => {
  let height = 0;
  for (let i = 0; i < props.data.length; i++) {
    height += heightCache.value.get(i) || props.estimatedItemHeight;
  }
  return height;
});

const visibleData = computed(() => {
  return props.data.slice(startIndex.value, endIndex.value);
});

const offset = computed(() => {
  let offset = 0;
  for (let i = 0; i < startIndex.value; i++) {
    offset += heightCache.value.get(i) || props.estimatedItemHeight;
  }
  return offset;
});

const bottomSpacerHeight = computed(() => {
  let height = 0;
  for (let i = endIndex.value; i < props.data.length; i++) {
    height += heightCache.value.get(i) || props.estimatedItemHeight;
  }
  return height;
});

function onScroll(event) {
  scrollTop.value = event.target.scrollTop;
  updateVisibleRange();
}

function updateVisibleRange() {
  // 二分查找起始索引
  let low = 0;
  let high = props.data.length - 1;
  
  while (low <= high) {
    const mid = Math.floor((low + high) / 2);
    let midOffset = 0;
    for (let i = 0; i < mid; i++) {
      midOffset += heightCache.value.get(i) || props.estimatedItemHeight;
    }
    
    if (midOffset === scrollTop.value) {
      startIndex.value = mid;
      break;
    } else if (midOffset < scrollTop.value) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }
  
  if (low > high) {
    startIndex.value = low;
  }
  
  // 计算结束索引
  let accumulatedHeight = 0;
  endIndex.value = startIndex.value;
  
  while (accumulatedHeight < props.containerHeight && endIndex.value < props.data.length) {
    accumulatedHeight += heightCache.value.get(endIndex.value) || props.estimatedItemHeight;
    endIndex.value++;
  }
}

function onItemResize(index, height) {
  heightCache.value.set(index, height);
  // 触发重新计算
  nextTick(() => {
    updateVisibleRange();
  });
}

onUpdated(() => {
  // 测量实际高度
  itemRefs.value.forEach((el, idx) => {
    if (el && el.offsetHeight) {
      const actualIndex = startIndex.value + idx;
      if (heightCache.value.get(actualIndex) !== el.offsetHeight) {
        onItemResize(actualIndex, el.offsetHeight);
      }
    }
  });
});
</script>

性能优化技巧 ​

1. 添加缓冲区 ​

在可视区域上下各增加几个项目作为缓冲，避免快速滚动时出现白屏：

javascript
const BUFFER_COUNT = 5; // 上下各缓冲 5 个项目

const startIndex = Math.max(0, calculatedStartIndex - BUFFER_COUNT);
const endIndex = Math.min(
  totalItems,
  calculatedEndIndex + BUFFER_COUNT
);

let rafId = null;

总结 ​

虚拟滚动的核心在于**"只渲染可见内容，用占位维持滚动"**，关键计算逻辑包括：

1. 可视范围计算（核心公式） ​

javascript
// 步骤1：计算起始索引 - 根据滚动位置确定第一个可见项
// Math.floor 确保即使部分可见也算作已滚动过
startIndex = Math.floor(scrollTop / itemHeight)

// 步骤2：计算可视数量 - 容器高度能容纳多少个项目
// Math.ceil 确保完全覆盖可视区域，避免底部留白
visibleCount = Math.ceil(containerHeight / itemHeight)

// 步骤3：计算结束索引 - 确定最后一个可见项（不超过总数）
// Math.min 防止索引越界
endIndex = Math.min(startIndex + visibleCount, totalItems)

3. 动态高度优化策略 ​

  • 预估高度：使用平均高度作为初始值
  • 二分查找：快速定位起始索引 O(log n)
  • 高度缓存：Map 存储实际测量高度，避免重复计算
  • 增量更新：只重新计算受影响区域的偏移量

4. 性能优化要点 ​

  • ✅ 添加缓冲区（Buffer）防止白屏
  • ✅ 使用 requestAnimationFrame 防抖
  • ✅ 优先使用 transform 而非 margin/top
  • ✅ 缓存计算结果，避免重复运算
  • ✅ 动态高度场景使用二分查找替代线性遍历

掌握这些核心逻辑后，可以根据业务需求灵活实现固定高度或动态高度的虚拟滚动方案。
